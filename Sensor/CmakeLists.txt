if(CMAKE_SIZEOF_VOID_P EQUAL 8)
        set(ARCH_SUFFIX "64")
else(CMAKE_SIZEOF_VOID_P EQUAL 4)
        set(ARCH_SUFFIX "32")
endif()

add_library(Sensor SHARED)

target_sources( Sensor
    PRIVATE

    FILE_SET HEADERS
    BASE_DIRS includes
    FILES
	includes/framework.h
	includes/pch.h
    	includes/hooking.h
)

target_include_directories(Sensor PRIVATE ${DETOURS_INCLUDE_DIRS})

target_sources( Sensor
    PRIVATE

    FILE_SET SOURCES
    FILES
    	exports.cpp
    	hooking.cpp
    	pch.cpp
)

target_compile_features(Sensor PRIVATE cxx_std_20)

target_link_libraries(Sensor PRIVATE
    spdlog::spdlog
    $<$<CONFIG:Debug>:${DETOURS_LIBRARY_DEBUG}>
    $<$<NOT:$<CONFIG:Debug>>:${DETOURS_LIBRARY_RELEASE}>
    phnt::phnt
    Userenv.lib
)

set_target_properties(Sensor PROPERTIES
    OUTPUT_NAME "Sensor${ARCH_SUFFIX}"
    LIBRARY_OUTPUT_DIRECTORY "${OUTPUT_DIR}"
)

target_sources( SensorStartup

    BASE_DIRS includes
    FILES

    	includes/hooking.h

    	hooking.cpp
    	pch.cpp

    phnt::phnt
    Userenv.lib
)

set_target_properties(SensorStartup PROPERTIES
    OUTPUT_NAME "SensorStartup${ARCH_SUFFIX}"
    LIBRARY_OUTPUT_DIRECTORY "${OUTPUT_DIR}"
)
